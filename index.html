async function handleSubmit() { const time = document.getElementById('feedTime').value; const type = document.querySelector('#typeGroup .active').dataset.val; const prepared = Number(document.getElementById('preparedAmount').value) || 0; const leftover = Number(document.getElementById('leftoverAmount').value) || 0; if (!time) return alert("請選擇餵奶時間"); const record = { id: currentEditId || 'rec_' + Date.now() + '_' + Math.random().toString(36).substr(2, 4), time, type, prepared, leftover, actual: Math.max(0, prepared - leftover) }; // 本地快照儲存 let records = getLocalRecords(); if (currentEditId) { records = records.map(r => r.id === currentEditId ? record : r); } else { records.push(record); } saveLocalRecords(records); // 重置表單 currentEditId = null; document.getElementById('leftoverAmount').value = ''; setNowTime(); // 向 GAS 發送更新（使用 text/plain 避開 CORS 預檢限制） if (GAS_URL) { try { const payload = JSON.stringify({ action: 'CREATE_OR_UPDATE', data: record }); await fetch(GAS_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: payload }); console.log("雲端同步成功"); } catch (err) { console.error("雲端發送失敗", err); } } } async function deleteRecord(id) { if (!confirm("確定要刪除這筆紀錄嗎？")) return; let records = getLocalRecords().filter(r => r.id !== id); saveLocalRecords(records); if (GAS_URL) { try { const payload = JSON.stringify({ action: 'DELETE', id }); await fetch(GAS_URL, { method: 'POST', headers: { 'Content-Type': 'text/plain;charset=utf-8' }, body: payload }); console.log("雲端刪除成功"); } catch (err) { console.error("雲端刪除失敗", err); } } }
